\tdsection{Why edge, why this benchmark}{%
  Poultry disease costs; lab diagnosis out of reach; the deploy gap.}
% =============================================================================

\begin{frame}{The gap this paper closes}

  \slidelead{Farm-level detection needs models that run on a Pi, not a GPU.}

  Poultry disease (Coccidiosis, Salmonellosis, Newcastle) causes major losses,
  yet lab diagnosis is out of reach for small farms. Prior work hits
  \acb{99.12\%} (EfficientNet-B0) and \acb{97.1\%} (MobileNetV2) — but targets
  the cloud, not the edge.

  Two questions drive this work: which lightweight architectures survive
  \code{INT8} on CPU, and what does the accuracy-latency tradeoff look like
  on a \acb{Raspberry Pi 5}?

  \aside{Scope: three models, one board, one runtime (ORT 1.29.0). No claimed
  generality beyond the measured stack.}

\end{frame}

\begin{frame}{The benchmark, and what was wrong with it}

  \slidelead{8,770 published images — audited, then deduplicated to 8,153.}

  \begin{stepflow}{3}
    \stepcard{claimed}{8,770 leak-free images (FecalFed benchmark)}
    \steparrow
    \stepcard{audit}{pHash audit: 617 near-duplicate pairs, 94 test images byte-identical to train}
    \steparrow
    \stepcard{used}{sterile 8,153 split, test n = 1,728, seed 42}
  \end{stepflow}

  \vspace{0.7em}
  {\small
  \acb{All reported numbers use the deduplicated split.} Four classes:
  Coccidiosis, Healthy, Salmonella, Newcastle (NCD underrepresented: 141 test).
  \par}

\end{frame}

  Dynamic INT8 keeps two models, destroys the third.}

\begin{frame}{Why (hypothesis): one shared scale, 127 levels}

  \slidelead{Per-tensor: the widest channel sets the scale for every channel.}

  \begin{center}
  \begin{tikzpicture}[x=0.0385cm, y=1cm]
    % per-tensor: widest channel
    \node[anchor=west, font=\footnotesize] at (-127,2.9)
      {Per-tensor: widest channel spans all 127 levels each side};
    \draw[line width=4pt, oliveGreen] (-127,2.55) -- (127,2.55);

    % per-tensor: narrow channel (112x smaller range => ~1.13 levels)
    \node[anchor=west, font=\footnotesize] at (-127,1.95)
      {Per-tensor: a channel with 112$\times$ smaller range gets $127/112 \approx 1$ level};
    \node[anchor=west, font=\footnotesize] at (-127,1.6)
      {$\Rightarrow$ quantized to a near-constant (channels at 0\,dB, audit JSON)};
    \draw[line width=4pt, red!70!black] (-1.13,1.25) -- (1.13,1.25);

    % per-channel: same narrow channel, own scale
    \node[anchor=west, font=\footnotesize] at (-127,0.6)
      {Per-channel: that channel gets its own scale and the full range back};
    \draw[line width=4pt, oliveGreen!70] (-127,0.25) -- (127,0.25);

    % axis
    \draw[oliveInk!50] (-127,-0.25) -- (127,-0.25);
    \foreach \t in {-127,0,127}
      \draw[oliveInk!60] (\t,-0.35) -- (\t,-0.15);
    \node[font=\scriptsize, below] at (-127,-0.35) {-127};
    \node[font=\scriptsize, below] at (0,-0.35) {0};
    \node[font=\scriptsize, below] at (127,-0.35) {+127};
  \end{tikzpicture}
  \end{center}

  {\small
  EB0's depthwise layers: median channel-range ratio \acb{39.94$\times$}
  (P95/P5), worst layer \acb{112$\times$}. MobileNetV2's median is only
  9.20$\times$ and ShuffleNetV2's 5.32$\times$, and both survive. Supported by SQNR and the family ablation; a single-layer ablation would test it.\par}

\end{frame}

\begin{frame}{Per-channel recovers it: 9.32\% vs 94.56\%}

  {\footnotesize
  \begin{tabular}{@{}>{\raggedright\arraybackslash}p{4.6cm}
                    >{\raggedright\arraybackslash}p{2.6cm}
                    >{\raggedright\arraybackslash}p{2.6cm}
                    >{\raggedright\arraybackslash}p{3.0cm}@{}}
    \toprule
    & \thead{acc} & \thead{size} & \thead{Pi t=4} \\
    \midrule
    \textbf{dynamic} & 10.42\% & 4.16 MB & 63.99 ms \\
    \textbf{static per-tensor} & 9.32\% & 4.29 MB & 13.63 ms \\
    \textbf{static per-channel} & \textbf{94.56\%} & 4.67 MB & \textbf{13.50 ms} \\
    \bottomrule
  \end{tabular}\par}

  \vspace{0.5em}
  {\small
  \acb{Recovery, not parity}: still 3.9 pp below FP32 (98.50\% on the ONNX graph).
  Balanced 25/class replication on a retrained model: per-tensor 27.14\% vs
  per-channel 93.40\%.\par}

  \aside{Caveat up front: calibration used 100 images, \emph{all Coccidiosis}
  (class-ordered loader). Diagnostic evidence, not a robustness claim.}

\end{frame}

\begin{frame}{Which layers matter: the selective ablation}

  \slidelead{Depthwise granularity decides collapse in this setup.}

  {\footnotesize
  \begin{tabular}{@{}>{\raggedright\arraybackslash}p{6.4cm}
                    >{\raggedright\arraybackslash}p{2.2cm}
                    >{\raggedright\arraybackslash}p{2.2cm}
                    >{\raggedright\arraybackslash}p{2.0cm}@{}}
    \toprule
    & \thead{size} & \thead{acc} & \thead{collapse} \\
    \midrule
    \textbf{C depthwise per-ch, rest per-tensor} & 4.33 MB & \textbf{93.46\%} & no \\
    \textbf{D all per-ch except depthwise} & 4.51 MB & 8.33\% & yes \\
    \textbf{E depthwise left in FP32} & 4.69 MB & 97.16\% & no \\
    \textbf{F SE left in FP32} & 5.91 MB & 12.44\% & yes \\
    \bottomrule
  \end{tabular}\par}

  \vspace{0.5em}
  {\small
  Ablation is at \emph{family} level, on EB0 only. Whether individual severe
  layers drive the collapse is a hypothesis; a single-layer ablation would test it.
  Depthwise sensitivity is known (Sheng 2018); our contribution is the case study
  plus runtime behavior.\par}

\end{frame}

\tdsection{What the Pi actually does}{%
  Dynamic INT8 slows down; static per-channel speeds up 2.34x.}
% =============================================================================

\begin{frame}{On the Pi: dynamic INT8 is slower than FP32}

  {\footnotesize
  \begin{tabular}{@{}>{\raggedright\arraybackslash}p{3.2cm}
                    >{\raggedright\arraybackslash}p{2.4cm}
                    >{\raggedright\arraybackslash}p{2.4cm}
                    >{\raggedright\arraybackslash}p{2.6cm}
                    >{\raggedright\arraybackslash}p{2.2cm}@{}}
    \toprule
    & \thead{FP32 t=4} & \thead{INT8 t=4} & \thead{scaling} & \thead{slowdown} \\
    \midrule
    \textbf{MNv2} & 14.23 ms & 39.83 ms & 1.10x & 2.80x \\
    \textbf{SNv2} & 6.19 ms & 11.95 ms & 1.14x & 1.93x \\
    \textbf{EB0} & 31.56 ms & 64.39 ms & 1.19x & 2.04x \\
    \bottomrule
  \end{tabular}\par}

  \vspace{0.5em}
  {\small
  FP32 scales 2.20--2.33x from 1 to 4 threads; INT8 manages 1.10--1.19x.
  Each Conv expands into DynamicQuantizeLinear + ConvInteger + Mul + Cast;
  quantization kernels add 20.7--29.2\% overhead.\par}

  \aside{Setup: Raspberry Pi 5, ONNX Runtime 1.29.0, CPU only, batch 1,
  model-only timing, 200 runs after 10 warmup.}

\end{frame}

\end{frame}

\begin{frame}{So which model do you ship?}

  \slidelead{INT8 here is a diagnostic result, not a shipping recommendation.}

  \begin{stepflow}{3}
    \stepcard{ship}{SNv2 FP32: 6.19 ms, 97.69\%}
    \steparrow
    \stepcard{most accurate}{EB0 FP32: 31.56 ms, 98.55\%}
    \steparrow
    \stepcard{dominated}{EB0 static per-ch: 13.50 ms, 94.56\%}
  \end{stepflow}

  \vspace{0.7em}
  {\small
  EB0 static per-channel loses to SNv2 FP32 on \acb{both} speed and accuracy, and
  to MNv2 FP32 (97.92\%) on accuracy at similar latency. FP32 accuracy spans only
  0.86 pp, so hardware and scheme should decide. INT8 shrinks the model 3.3--3.7x
  but RAM at inference barely moves.\par}

  \aside{Limitations: single Coccidiosis-only calibration slice; one seed;
  energy not measured.}

\end{frame}

\tdsection{Close}{%
  Deployment depends on architecture, runtime, and granularity jointly.}
% =============================================================================

\begin{frame}{Takeaway and artifact}

  \slidelead{INT8 is not one thing: scheme $\times$ granularity $\times$ runtime decides.}

  \srcline{\textbf{Dynamic INT8} preserves MNv2/SNv2, collapses EB0 to 10.42\%}
  \srcline{\textbf{Per-channel static} recovers EB0 to 94.56\% and is 2.34x faster than FP32 — but still loses to SNv2 FP32}
  \srcline{\textbf{Depthwise granularity} decides collapse in this setup (93.46\% vs 8.33\%)}
  \srcline{Artifact: \code{github.com/jaweed3/paper\_poultry\_p2} + Zenodo 10.5281/zenodo.23374216}

  \aside{Thank you. Questions welcome — the failure modes are in the paper.}

\end{frame}
